- let pool; // declare a pool reference to share after initialization.

- await connection.end(); // close the temporary connection.
- pool = mysql.createPool(config.db); // build a connection pool for all queries.
- const createTableQuery = `...`; // define SQL for the users table with id, username, password, created_at.
- await pool.query(createTableQuery); // create the users table if missing.
- export function getPool() { if (!pool) throw new Error("Database not initialized. Call initializeDatabase() first."); return pool; } // return the pool once initialized.

4) models/userModel.js (new file)
- import { getPool } from "../database/db.js"; // import the initialized pool accessor.
- export async function findUserByUsername(username) { const pool = getPool(); const [rows] = await pool.execute("SELECT id, username, password FROM users WHERE username = ?", [username]); return rows[0]; } // query the users table for a matching username.
- export async function createUser(username, hashedPassword) { const pool = getPool(); const [result] = await pool.execute("INSERT INTO users (username, password) VALUES (?, ?)", [username, hashedPassword]); return result.insertId; } // insert a new user record securely.

5) controllers/authController.js (new file)
- import bcrypt from "bcrypt"; // use bcrypt to hash and verify passwords.
- import { findUserByUsername, createUser } from "../models/userModel.js"; // import user database helpers.
- const SALT_ROUNDS = 10; // define bcrypt work factor.
- export async function signup(req, res, next) { ... } // handle signup logic.
- const { username, password, confirmPassword } = req.body; // pull signup values from request body.
- const trimmedUsername = username?.trim(); // normalize the username input.
- validate username, password, and confirmPassword, returning errors for empty values or mismatched passwords. // ensure input is correct.
- const existingUser = await findUserByUsername(trimmedUsername); // check duplicate username.
- if (existingUser) return res.status(409).json({ success: false, message: "Username already exists." }); // return duplicate error.
- const hashedPassword = await bcrypt.hash(password, SALT_ROUNDS); // hash the password.
- await createUser(trimmedUsername, hashedPassword); // save the new user.
- return res.status(201).json({ success: true, message: "Sign up successful." }); // return signup success.
- export async function login(req, res, next) { ... } // handle login logic.
- const { username, password } = req.body; // pull login values.
- const trimmedUsername = username?.trim(); // normalize username.
- validate username and password, returning errors for empty values. // ensure required login fields are present.
- const user = await findUserByUsername(trimmedUsername); // fetch stored user.
- if (!user) return res.status(404).json({ success: false, message: "User does not exist." }); // return missing user error.
- const isPasswordValid = await bcrypt.compare(password, user.password); // compare password with stored hash.
- if (!isPasswordValid) return res.status(401).json({ success: false, message: "Incorrect password." }); // return wrong password error.
- return res.status(200).json({ success: true, message: "Login successful" }); // return login success.

6) routes/authRoutes.js (new file)
- import express from "express"; // load Express router helper.
- import { signup, login } from "../controllers/authController.js"; // import signup and login controller functions.
- const router = express.Router(); // create a router instance.
- router.post("/signup", signup); // map POST /signup to signup controller.
- router.post("/login", login); // map POST /login to login controller.
- export default router; // export the router for use in server.js.
